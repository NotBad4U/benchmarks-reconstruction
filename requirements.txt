# Install into a local venv, never globally:
#
#     python3 -m venv .venv
#     ./.venv/bin/pip install -r requirements.txt
#
# collects.py uses the standard library alone; numpy, pandas, tabulate and
# python-dateutil were needed only by the joblog parsers since removed.
doit>=0.36

# `doit graph`: writes the task dependency graph as a Graphviz .dot file.
#     ./.venv/bin/doit graph --show-subtasks -o tasks.dot
#     dot -Tpng tasks.dot -o tasks.png
# Pulls in pygraphviz, a C extension.  PyPI has prebuilt manylinux_2_28 and
# macOS wheels for it, so no compiler is needed -- but on a Linux host with
# glibc older than 2.28, pip falls back to building from source and needs the
# Graphviz headers installed.  Rendering the .png needs the `dot` binary.
doit-graph>=0.3

# Non-Python tools dodo.py shells out to: cvc5, carcara, lambdapi, hyperfine.
# On the ITU cluster, slurm/setup.job builds all of them into $HOME.
#
# `doit report`: BenchExec's table-generator, which turns the status records
# into a sortable HTML table with quantile and scatter plots.  It is pure Python
# and works on macOS.  (BenchExec's *runexec* is the Linux-only part -- it loads
# libc.so.6 and needs cgroup v2 -- and this pipeline does not use it.)
BenchExec>=3.30
